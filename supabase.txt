-- Function to calculate rent differential percentage
CREATE OR REPLACE FUNCTION calculate_rent_differential()
  RETURNS TRIGGER
  LANGUAGE plpgsql
AS $$
BEGIN
  -- Only calculate if we have both original and sublease rent
  IF NEW.original_lease_rent IS NOT NULL AND NEW.price IS NOT NULL THEN
    -- Calculate difference percentage
    NEW.rent_differential := (
      (NEW.price - NEW.original_lease_rent) / NEW.original_lease_rent * 100
    )::numeric(5,2);
    
    -- Calculate lease term in months if dates are provided
    IF NEW.sublease_from IS NOT NULL AND NEW.sublease_to IS NOT NULL THEN
      NEW.original_lease_term := 
        EXTRACT(YEAR FROM age(NEW.sublease_to, NEW.sublease_from)) * 12 +
        EXTRACT(MONTH FROM age(NEW.sublease_to, NEW.sublease_from));
    END IF;
  END IF;
  
  RETURN NEW;
END;
$$;

-- Create trigger to automatically update rent differential
CREATE TRIGGER update_rent_differential
  BEFORE INSERT OR UPDATE ON properties
  FOR EACH ROW
  EXECUTE FUNCTION calculate_rent_differential();

-- Set up functions for event capacity checks

-- Function to check if an event has available capacity before a user can join
CREATE OR REPLACE FUNCTION check_event_capacity()
RETURNS TRIGGER AS $$
DECLARE
  event_capacity INTEGER;
  current_participants INTEGER;
BEGIN
  -- Get the event capacity
  SELECT capacity INTO event_capacity
  FROM events
  WHERE id = NEW.event_id;
  
  -- Count current participants
  SELECT COUNT(*) INTO current_participants
  FROM event_participants
  WHERE event_id = NEW.event_id;
  
  -- Check if adding this participant would exceed capacity
  IF current_participants >= event_capacity THEN
    RAISE EXCEPTION 'Event is at full capacity';
  END IF;
  
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Create trigger to check capacity before inserting a new participant
CREATE TRIGGER check_event_capacity_trigger
BEFORE INSERT ON event_participants
FOR EACH ROW
EXECUTE FUNCTION check_event_capacity();

-- Add function to prevent users from being added to groups with block relationships
CREATE OR REPLACE FUNCTION check_group_blocks() RETURNS TRIGGER AS $$
BEGIN
  -- Check if the new participant has any blocking relationships with existing group members
  IF EXISTS (
    SELECT 1 FROM user_blocks b
    JOIN chat_participants p ON 
      (b.blocker_id = p.profile_id OR b.blocked_id = p.profile_id)
    WHERE 
      p.chat_id = NEW.chat_id
      AND (
        (b.blocker_id = NEW.profile_id AND b.blocked_id = p.profile_id)
        OR
        (b.blocked_id = NEW.profile_id AND b.blocker_id = p.profile_id)
      )
  ) THEN
    -- This check will be handled in the application logic instead
    -- to provide better user feedback
    -- RAISE EXCEPTION 'Cannot add user to group due to block relationships';
    -- We'll allow the insert but log a warning
    RAISE WARNING 'User % being added to group % despite block relationships', NEW.profile_id, NEW.chat_id;
  END IF;
  
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Add trigger for the above function

CREATE TRIGGER check_group_blocks_trigger

-- Create a function to get a user's groups
CREATE OR REPLACE FUNCTION get_user_groups(user_id UUID)
RETURNS TABLE (
  chat_id UUID,
  group_name TEXT,
  is_admin BOOLEAN,
  member_count BIGINT,
  last_message TEXT,
  last_message_at TIMESTAMPTZ
) AS $$
BEGIN
  RETURN QUERY
  SELECT
    c.id,
    c.group_name,
    p.role = 'admin' AS is_admin,
    COUNT(p2.profile_id) AS member_count,
    c.last_message,
    c.last_message_at
  FROM
    chat_rooms c
    JOIN chat_participants p ON c.id = p.chat_id AND p.profile_id = user_id
    JOIN chat_participants p2 ON c.id = p2.chat_id
  WHERE
    c.type = 'group'
  GROUP BY
    c.id, c.group_name, p.role, c.last_message, c.last_message_at
  ORDER BY
    c.last_message_at DESC NULLS LAST;
END;
$$ LANGUAGE plpgsql;

-- Create a function to add a system message to a chat
CREATE OR REPLACE FUNCTION add_system_message(
  p_chat_id UUID,
  p_content TEXT
) RETURNS UUID AS $$
DECLARE
  message_id UUID;
BEGIN
  -- Insert the system message
  INSERT INTO messages (
    chat_id,
    sender_id,
    content,
    created_at,
    is_system_message,
    read_by
  )
  VALUES (
    p_chat_id,
    NULL, -- system messages have no sender
    p_content,
    NOW(),
    TRUE,
    ARRAY[]::UUID[]
  )
  RETURNING id INTO message_id;
  
  -- Update the chat's last message
  UPDATE chat_rooms
  SET
    last_message = p_content,
    last_message_at = NOW()
  WHERE
    id = p_chat_id;
    
  RETURN message_id;
END;
$$ LANGUAGE plpgsql;

-- Create a trigger function to add system messages on participant changes
CREATE OR REPLACE FUNCTION on_chat_participant_change() RETURNS TRIGGER AS $$
DECLARE
  chat_type TEXT;
  user_name TEXT;
  adder_name TEXT;
  message_text TEXT;
BEGIN
  -- Get chat type
  SELECT type INTO chat_type FROM chat_rooms WHERE id = 
    CASE
      WHEN TG_OP = 'DELETE' THEN OLD.chat_id
      ELSE NEW.chat_id
    END;
  
  -- Only proceed for group chats
  IF chat_type != 'group' THEN
    RETURN NULL;
  END IF;
  
  IF TG_OP = 'INSERT' THEN
    -- Get user name
    SELECT full_name INTO user_name FROM profiles WHERE id = NEW.profile_id;
    
    -- Get adder name if available
    IF NEW.added_by IS NOT NULL THEN
      SELECT full_name INTO adder_name FROM profiles WHERE id = NEW.added_by;
      message_text := user_name || ' was added by ' || adder_name;
    ELSE
      message_text := user_name || ' joined the group';
    END IF;
    
    -- Add system message
    PERFORM add_system_message(NEW.chat_id, message_text);
    
  ELSIF TG_OP = 'DELETE' THEN
    -- Get user name
    SELECT full_name INTO user_name FROM profiles WHERE id = OLD.profile_id;
    
    -- Determine if user left or was removed
    -- This is a simplification - app logic can provide more context
    message_text := user_name || ' left the group';
    
    -- Add system message
    PERFORM add_system_message(OLD.chat_id, message_text);
  END IF;
  
  RETURN NULL;
END;
$$ LANGUAGE plpgsql;
